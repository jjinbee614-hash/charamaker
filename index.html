Signup bonus: 500 → 200 credits (1 character + 1 image trial) -- 2. Gallery approval bonus: per-type -- - characters / images → +20 credits (unchanged) -- - videos → +100 credits (videos cost more to make + are higher -- marketing value, so the bonus is bigger) -- ═══════════════════════════════════════════════════════════ -- 1) New signup grant: 200 credits (was 500) create or replace function public.handle_new_user_credits() returns trigger language plpgsql security definer as $$ begin insert into user_credits (user_id, credits, plan_name, signup_bonus_given) values (new.id, 200, 'free', true) on conflict (user_id) do nothing; insert into credit_transactions (user_id, delta, reason) values (new.id, 200, 'signup'); return new; end; $$; -- 2) Gallery approval: variable bonus per content type -- Approval triggers fire on characters / images / videos. We use TG_TABLE_NAME -- to pick the right amount. create or replace function public.handle_gallery_approval() returns trigger language plpgsql security definer as $$ declare v_already_granted boolean; v_bonus int; begin -- Only fire when transitioning to 'approved' if new.gallery_status = 'approved' and (old.gallery_status is null or old.gallery_status != 'approved') then -- Check if we already granted credits for this row (idempotency) select exists ( select 1 from credit_transactions where ref_id = new.id and reason = 'gallery_approved' ) into v_already_granted; if not v_already_granted and new.user_id is not null then -- Bigger bonus for videos (higher production cost + marketing value) v_bonus := case TG_TABLE_NAME when 'videos' then 100 else 20 -- characters and images end; perform public.grant_credits(new.user_id, v_bonus, 'gallery_approved', new.id); end if; end if; return new; end; $$; -- Triggers stay the same — they reference handle_gallery_approval -- which now branches on TG_TABLE_NAME. -- ═══════════════════════════════════════════════════════════ -- Existing users keep whatever balance they have. -- New signups starting now will get 200 credits instead of 500. -- Existing pending gallery rows get the new amount when approved. -- ═══════════════════════════════════════════════════════════
